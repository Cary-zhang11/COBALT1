# SkillFlow v1.0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the SkillFlow v1.0 MVP — a full-stack Next.js app where users upload requirements, AI matches and recommends Skills, and LangGraph executes workflows with pause/resume checkpoints, generating markdown outputs.

**Architecture:** Next.js 14 App Router with API Routes handling synchronous LangGraph execution. PostgresSaver persists workflow checkpoints to PostgreSQL. Frontend uses 5-second polling against `/api/projects/:id/status`. Four built-in Skills are hardcoded with LLM prompt templates.

**Tech Stack:** Next.js 14, React 18, TypeScript, Tailwind CSS, Prisma, PostgreSQL, LangGraph + PostgresSaver, Vercel AI SDK, mammoth, pdf-parse, js-yaml

```typescript
// parsers/index.ts
import { parseText } from "./text";
import { parseDocx } from "./docx";
import { parsePdf } from "./pdf";
import { parseJsonYaml } from "./json-yaml";

const UPLOAD_CONFIG = {
  maxFileSize: 5 * 1024 * 1024,
  maxTextLength: 50_000,
  allowedExtensions: [".md", ".txt", ".docx", ".pdf", ".json", ".yaml", ".yml"],
};

export { UPLOAD_CONFIG };

export function getExtension(filename: string): string {
  return filename.slice(filename.lastIndexOf(".")).toLowerCase();
}

export async function parseFile(filename: string, buffer: Buffer): Promise<string> {
  const ext = getExtension(filename);

  if (!UPLOAD_CONFIG.allowedExtensions.includes(ext)) {
    throw new Error(`Unsupported file format: ${ext}. Allowed: ${UPLOAD_CONFIG.allowedExtensions.join(", ")}`);
  }

  let text: string;
  switch (ext) {
    case ".md":
    case ".txt":
      text = await parseText(buffer);
      break;
    case ".docx":
      text = await parseDocx(buffer);
      break;
    case ".pdf":
      text = await parsePdf(buffer);
      break;
    case ".json":
    case ".yaml":
    case ".yml":
      text = await parseJsonYaml(buffer);
      break;
    default:
      throw new Error(`No parser for extension: ${ext}`);
  }

  if (text.length > UPLOAD_CONFIG.maxTextLength) {
    throw new Error(`Text exceeds ${UPLOAD_CONFIG.maxTextLength} characters limit. Current: ${text.length}`);
  }

  return text;
}
```

### Step 3.2: Write upload API

```typescript
// app/api/projects/[id]/upload/route.ts
import { NextRequest, NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";
import { parseFile, UPLOAD_CONFIG } from "@/parsers";

export async function POST(
  req: NextRequest,
  { params }: { params: { id: string } }
) {
  const formData = await req.formData();
  const file = formData.get("file") as File | null;

  if (!file) {
    return NextResponse.json({ error: "No file provided" }, { status: 400 });
  }

  if (file.size > UPLOAD_CONFIG.maxFileSize) {
    return NextResponse.json(
      { error: `File exceeds 5MB limit. Size: ${(file.size / 1024 / 1024).toFixed(2)}MB` },
      { status: 413 }
    );
  }

  const buffer = Buffer.from(await file.arrayBuffer());

  try {
    const requirement = await parseFile(file.name, buffer);

    const project = await prisma.project.update({
      where: { id: params.id },
      data: {
        requirement,
        fileName: file.name,
        fileSize: file.size,
      },
    });

    return NextResponse.json({ project, requirementLength: requirement.length });
  } catch (err) {
    const message = err instanceof Error ? err.message : "Parse failed";
    return NextResponse.json({ error: message }, { status: 400 });
  }
}
```

### Step 6.1: Write skill matcher

```typescript
// skills/matcher.ts
import { callLLM } from "@/lib/ai";
import { BUILT_IN_SKILLS } from "./registry";
import { SkillRecommendation } from "./types";

export async function matchSkills(requirementText: string): Promise<SkillRecommendation[]> {
  const prompt = `根据以下需求内容，从给定的 Skills 中推荐最匹配的 3 个。

需求内容：
${requirementText.slice(0, 8000)}

可选 Skills：
${BUILT_IN_SKILLS.map((s) => `- ${s.name}: ${s.description}`).join("\n")}

请以 JSON 格式返回推荐结果，格式如下：
[
  { "skillId": "...", "matchScore": 0.95, "matchReason": "..." }
]

只返回 JSON 数组，不要其他内容。`;

  const response = await callLLM(prompt);

  try {
    const jsonMatch = response.match(/\[[\s\S]*\]/);
    const parsed: SkillRecommendation[] = jsonMatch ? JSON.parse(jsonMatch[0]) : JSON.parse(response);
    return parsed.slice(0, 3).map((r) => ({
      ...r,
      name: BUILT_IN_SKILLS.find((s) => s.id === r.skillId)?.name || r.skillId,
    }));
  } catch {
    // Fallback: simple keyword matching
    return BUILT_IN_SKILLS.slice(0, 3).map((s, i) => ({
      skillId: s.id,
      name: s.name,
      matchScore: 0.9 - i * 0.1,
      matchReason: "基于关键词匹配",
    }));
  }
}
```

```typescript
// workflow/nodes.ts
import { WorkflowState } from "./types";
import { matchSkills } from "@/skills/matcher";
import { callLLMWithSystem } from "@/lib/ai";
import { getSkillById } from "@/skills/registry";
import { WorkflowMessage } from "@/skills/types";

export async function parseNode(state: typeof WorkflowState.State) {
  return {
    currentStep: "match",
    messages: [
      {
        role: "ai" as const,
        content: "解析完成，提取需求文本",
        type: "status",
      },
    ],
  };
}

export async function matchNode(state: typeof WorkflowState.State) {
  const recommendations = await matchSkills(state.requirementText);
  return {
    matchedSkills: recommendations,
    currentStep: "waitSkillSelect",
    messages: [
      {
        role: "ai" as const,
        content: `基于需求分析，为您匹配了以下 Skills：${recommendations.map((r) => r.name).join(", ")}`,
        type: "status",
      },
    ],
  };
}

export function pauseNode(message: string) {
  return async (state: typeof WorkflowState.State) => {
    return {
      status: "paused" as const,
      messages: [
        {
          role: "ai" as const,
          content: message,
          type: "pause",
        },
      ],
    };
  };
}

export async function clarifyNode(state: typeof WorkflowState.State) {
  const skill = getSkillById(state.selectedSkillId ?? "");
  if (!skill) {
    return {
      status: "failed" as const,
      messages: [{ role: "ai" as const, content: "Skill not found", type: "error" }],
    };
  }

  const prompt = `根据以下需求，生成 1-2 个简短的澄清问题，帮助更精确地生成${skill.name}产出。

需求：
${state.requirementText.slice(0, 5000)}

请以 JSON 数组格式返回问题列表：["问题1", "问题2"]`;

  const response = await callLLMWithSystem(
    "你是一个需求分析助手。只返回 JSON 数组格式的问题列表。",
    prompt
  );

  let questions: string[] = [];
  try {
    const match = response.match(/\[[\s\S]*\]/);
    questions = match ? JSON.parse(match[0]) : JSON.parse(response);
  } catch {
    questions = ["是否有特定的技术栈或框架需要考虑？"];
  }

  return {
    currentStep: "waitClarify",
    messages: [
      {
        role: "ai" as const,
        content: `在正式开始${skill.name}之前，我需要确认几个细节：\n\n${questions.map((q, i) => `${i + 1}. ${q}`).join("\n")}`,
        type: "clarify",
      },
    ],
  };
}

export async function generateNode(state: typeof WorkflowState.State) {
  const skill = getSkillById(state.selectedSkillId ?? "");
  if (!skill) {
    return {
      status: "failed" as const,
      messages: [{ role: "ai" as const, content: "Skill not found", type: "error" }],
    };
  }

  // Load prompt template by skill id
  const { buildPrompt } = await import(`@/skills/prompts/${skill.id}`);
  const prompt = buildPrompt(state.requirementText, state.messages);

  const output = await callLLMWithSystem(
    `你是一个专业的${skill.name}专家。请根据需求生成高质量的${skill.name}产出，使用 Markdown 格式。`,
    prompt
  );

  return {
    output,
    currentStep: "waitReview",
    messages: [
      {
        role: "ai" as const,
        content: `${skill.name}初稿已完成，请审核以下内容：\n\n---\n${output.slice(0, 500)}...\n\n---\n\n如果满意请点击「通过」，或输入修改意见。`,
        type: "review",
      },
    ],
  };
}

export async function outputNode(state: typeof WorkflowState.State) {
  return {
    status: "completed" as const,
    currentStep: "output",
    messages: [
      {
        role: "ai" as const,
        content: "产出已完成",

      },

  };
}
```

### Step 7.3: Write graph definition

```typescript
// workflow/graph.ts
import { StateGraph } from "@langchain/langgraph";
import { PostgresSaver } from "@langchain/langgraph-checkpoint-postgres";
import { WorkflowState } from "./types";
import {
  parseNode,
  matchNode,
  pauseNode,
  clarifyNode,
  generateNode,
  outputNode,
} from "./nodes";

function createCheckpointer() {
  return new PostgresSaver({
    connectionString: process.env.DATABASE_URL!,
  });
}

export function createWorkflowGraph() {
  const graph = new StateGraph(WorkflowState)
    .addNode("parse", parseNode)
    .addNode("match", matchNode)
    .addNode("waitSkillSelect", pauseNode("请选择一个 Skill 开始执行"))
    .addNode("clarify", clarifyNode)
    .addNode("waitClarify", pauseNode("请回答澄清问题"))
    .addNode("generate", generateNode)
    .addNode("waitReview", pauseNode("请审核生成的内容"))
    .addNode("output", outputNode)
    .addEdge("__start__", "parse")
    .addEdge("parse", "match")
    .addEdge("match", "waitSkillSelect")
    .addEdge("waitSkillSelect", "clarify")
    .addEdge("clarify", "generate")
    .addEdge("generate", "waitReview")
    .addEdge("waitReview", "output")
    .addEdge("output", "__end__");

  const checkpointer = createCheckpointer();
  return graph.compile({ checkpointer });
}
```

```typescript
// app/api/projects/[id]/start-workflow/route.ts
import { NextRequest, NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";
import { createWorkflowGraph } from "@/workflow/graph";

export async function POST(
  req: NextRequest,
  { params }: { params: { id: string } }
) {
  const body = await req.json();
  const { skillId } = body;

  if (!skillId) {
    return NextResponse.json({ error: "skillId is required" }, { status: 400 });
  }

  const project = await prisma.project.findUnique({
    where: { id: params.id },
  });

  if (!project) {
    return NextResponse.json({ error: "Project not found" }, { status: 404 });
  }

  if (!project.requirement) {
    return NextResponse.json(
      { error: "Project has no requirement text" },
      { status: 400 }
    );
  }

  // Initialize or reset execution record
  await prisma.skillExecution.upsert({
    where: { projectId: params.id },
    create: {
      projectId: params.id,
      skillId,
      status: "running",
      currentStep: "parse",
      messages: [],
    },
    update: {
      skillId,
      status: "running",
      currentStep: "parse",
      messages: [],
      output: null,
      completedAt: null,
    },
  });

  await prisma.project.update({
    where: { id: params.id },
    data: { status: "running" },
  });

  const app = createWorkflowGraph();

  try {
    const result = await app.invoke(
      {
        projectId: params.id,
        requirementText: project.requirement,
        selectedSkillId: null,
        currentStep: "parse",
        messages: [],
        output: null,
        status: "running",
      },
      { configurable: { thread_id: params.id } }
    );

    // Sync execution state to DB
    await prisma.skillExecution.update({
      where: { projectId: params.id },
      data: {
        status: result.status,
        currentStep: result.currentStep,
        messages: result.messages as any,
        output: result.output,
        completedAt: result.status === "completed" ? new Date() : null,
      },
    });

    await prisma.project.update({
      where: { id: params.id },
      data: { status: result.status },
    });

    return NextResponse.json({
      status: result.status,
      currentStep: result.currentStep,
      messages: result.messages,
      output: result.output,
    });
  } catch (err) {
    const message = err instanceof Error ? err.message : "Workflow failed";
    await prisma.skillExecution.update({
      where: { projectId: params.id },
      data: { status: "failed" },
    });
    await prisma.project.update({
      where: { id: params.id },
      data: { status: "failed" },
    });
    return NextResponse.json({ error: message, status: "failed" }, { status: 500 });
  }
}
```

```typescript
// app/api/projects/[id]/resume/route.ts
import { NextRequest, NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";
import { createWorkflowGraph } from "@/workflow/graph";

export async function POST(
  req: NextRequest,
  { params }: { params: { id: string } }
) {
  const body = await req.json();
  const { answer, skillId } = body;

  const project = await prisma.project.findUnique({
    where: { id: params.id },
    include: { execution: true },
  });

  if (!project) {
    return NextResponse.json({ error: "Project not found" }, { status: 404 });
  }

  const execution = project.execution;
  if (!execution) {
    return NextResponse.json({ error: "No active execution" }, { status: 400 });
  }

  const app = createWorkflowGraph();

  // Build state update based on current step
  const currentMessages = (execution.messages as any[]) ?? [];
  const newMessages = answer
    ? [...currentMessages, { role: "user" as const, content: answer }]
    : currentMessages;

  const stateUpdate: Record<string, any> = {
    messages: newMessages,
    status: "running",
  };

  // If resuming from skill selection, set selectedSkillId
  if (execution.currentStep === "waitSkillSelect" && skillId) {
    stateUpdate.selectedSkillId = skillId;
  }

  await prisma.skillExecution.update({
    where: { projectId: params.id },
    data: { status: "running" },
  });

  await prisma.project.update({
    where: { id: params.id },
    data: { status: "running" },
  });

  try {
    const result = await app.invoke(stateUpdate, {
      configurable: { thread_id: params.id },
    });

    await prisma.skillExecution.update({
      where: { projectId: params.id },
      data: {
        status: result.status,
        currentStep: result.currentStep,
        messages: result.messages as any,
        output: result.output,
        completedAt: result.status === "completed" ? new Date() : null,
      },
    });

    await prisma.project.update({
      where: { id: params.id },
      data: { status: result.status },
    });

    return NextResponse.json({
      status: result.status,
      currentStep: result.currentStep,
      messages: result.messages,
      output: result.output,
    });
  } catch (err) {
    const message = err instanceof Error ? err.message : "Resume failed";
    await prisma.skillExecution.update({
      where: { projectId: params.id },
      data: { status: "failed" },
    });
    await prisma.project.update({
      where: { id: params.id },
      data: { status: "failed" },
    });
    return NextResponse.json({ error: message, status: "failed" }, { status: 500 });
  }
}
```
